\begin{frame}{The Denoising score link}
  \vspace{-100pt}
  \begin{lemma}[\cite{vincent2011connection}]
    \begin{equation}
      \PE{}{\state_0 | \state_{\sigma}=\lstate} = \lstate + \sigma^2 \nabla \log p_{\sigma}(\lstate) \eqsp,
  \end{equation}
  where $p_{\sigma} = \operatorname{Law}(\state_\sigma)$.  
  \end{lemma}
\end{frame}
\begin{frame}{First solution \citep{song2019generative}}
    $0 < \sigma_1 < \cdots < \sigma_T$.
    \begin{equation}
        \param \in \arg\min \sum_{t=1}^{N} \gamma_t^2 \PE{}{\|D_{\param}(\state_0 + \sigma_t Z, \sigma_t) - \state_0\|^2}
    \end{equation}
    \pause

    Sequential Langevin Algorithm: 
    \vspace{10pt}
    
    $\state_{T} \sim \gauss(0, \sigma_T^2\idm)$, define $s_{\param}(\lstate, \sigma) = \sigma^{-2}(D_{\param}(\lstate, \sigma) - \lstate)$.
    \vspace{10pt}
    \pause

    For $t=T-1, \cdots, 1$
    \begin{itemize}
      \item $\state_{0, t} = \state_{K, t+1}$
      \item $\state_{k, t} = \state_{k-1, t} + \eta s_{\param}(\state_{k-1, t}, \sigma) + \sqrt{2 \eta} Z_{k, t}$. 
    \end{itemize}
\end{frame}
\begin{frame}{Linking the noising procedure: \cite{ho2020denoising}}
  Consider the following Markov chain:
  \begin{enumerate}
    \item $\state_0 \sim \pdata$,
    \item $\state_t \sim \state_{t-1} + \sqrt{\sigma_t^2 - \sigma_{t-1}^2}$.
  \end{enumerate}

  \pause
  The joint law of this chain is
  \begin{equation}
     p_{0:T}(\lstate_{0:T}) = \pdata(\lstate_0) \prod_{t=1}^{T} \overbrace{\gauss(\lstate_{t}; \lstate_{t-1}, (\sigma_t^2 - \sigma_{t-1}^2) \idm)}^{^{\eqdef p_{t|t-1}(\lstate_t|\lstate_{t-1})}} \eqsp.
  \end{equation}
  
  \pause
  It holds that:
  \begin{enumerate}
    \item $p_{t|0}(\lstate_{t} | \lstate_0) = \gauss(\lstate_0, \sigma_t^2 \idm)$,
    \item $p_{t|0, t+1}(\lstate_{t}| \lstate_0, \lstate_{t+1}) = \gauss(\lstate_{t}, \lstate_0 + \frac{\sigma_t²}{\sigma_{t+1}^2}(\lstate_{t+1} - \lstate_0), \frac{\sigma_t^2}{\sigma_{t+1}^2}(\sigma_{t+1}^2 - \sigma_{t}^2)\idm)$
  \end{enumerate}

\end{frame}
\begin{frame}{Variational approximation}
  We want to reverse the noising process. Consider
  \begin{equation}
    q_{0:T}^{\param}(\lstate_{0:T}) \eqdef \gauss(\lstate_T; \mu_{T}, \eta_T^2 \idm) \prod_{t=0}^{T-1} \gauss(\lstate_{t}; \mu_{t, \param}(\lstate_{t+1}), \eta_{t+1}^2 \idm) \eqsp.
  \end{equation}
  \pause
  Our goal is to
  \begin{equation}
    {\arg\min}_{\param, \eta_{0:T}^2, \mu_{T}} \kl{p_{0:T}}{q_{0:T}^{\param}} \eqsp.
  \end{equation}

  \pause
  Before advancing, note that $p_{0:T}$ (as it is a Markov chain) admits the following decomposition
  $$p_{0:T}(\lstate_{0:T}) = p_T(\lstate_t) \prod_{t=0}^{T-1} p_{t|t+1}(\lstate_{t} | \lstate_{t+1}) \eqsp.$$

  Unfortunately, each $p_{t|t+1}$ is intractable for general $\pdata$.
\end{frame}
\begin{frame}{Calculations}
  \begin{align*}
    &\kl{p_{0:T}}{q_{0:T}^{\param}} = \\
    &\quad \PE{p_{0:T}}{\log\left(\frac{p_T(\lstate_t) \prod_{t=0}^{T-1} p_{t|t+1}(\lstate_{t} | \lstate_{t+1})}{\gauss(\lstate_T; \mu_{T}, \eta_T^2 \idm) \prod_{t=0}^{T-1} \gauss(\lstate_{t}; \mu_{t, \param}(\lstate_{t+1}), \eta_{t+1}^2 \idm)}\right)} \\
    &\quad =\kl{p_T}{\gauss(\mu_{T}, \eta_T^2 \idm)} \\
    &\quad + \sum_{t=1}^{T-1} \PE{p_{0:T}}{\kl{p_{t|t+1}(\cdot | \state_{t+1})}{\gauss(\mu_{t, \param}(\state_{t+1}), \eta_{t+1}^2 \idm)}} \\
    &\quad + \PE{p_{0:T}}{\kl{\pdata}{\gauss(\mu_{0, \param}(\state_{1}), \eta_{t+1}^2)}} \eqsp.
  \end{align*}

  \pause
  We now use the following property:
  \begin{lemma}
      If $\lambda \in \pset{2}$, then 
      $$ (\mu_\lambda, \Sigma_\lambda) \in \arg\min \kl{\lambda}{\gauss(\mu, \Sigma)} \eqsp,$$
      where $\mu_\lambda = \PE{\lambda}{\state}$ and $\Sigma = \PV{\lambda}{\state} = \PE{\lambda}{(\state - \mu_{\lambda})(\state - \mu_{\lambda})^T}$.
  \end{lemma}
\end{frame}

\begin{frame}{Backward Sampling}
    Algorithm:
    \begin{enumerate}
      \item Sample $\state_T \sim \gauss(0, \sigma_T^2)$,
      \item Sample for $t = T-1, \cdots, 1$,
    \end{enumerate}
    $$\state_{t} \sim \gauss\left(D_{\param}(\state_{t+1}, \sigma_{t+1}) + \frac{\sigma_t^2}{\sigma_{t+1}^2}(\state_{t+1} - D_{\param}(\state_{t+1}, \sigma_{t+1})),  \frac{\sigma_t^2}{\sigma_{t+1}^2}(\sigma_{t+1}^2 - \sigma_{t}^2)\idm\right) \eqsp.$$
\end{frame}
\begin{frame}{Variance Exploding vs Variance preserving}
  Often in the literature, you see the following noising process for $\beta_t \in (0, 1)$:
  \begin{equation}
    Y_t = \sqrt{1 - \beta_t} Y_{t-1} + \sqrt{\beta_t} Z_{t} \eqsp.
  \end{equation}

  It has the nice property that $\PE{}{Y_t} = 0$ and $\PV{}{Y_t} = 1$ (Thus variance preserving) and that
  $$p(y_t | y_0) = \gauss(\sqrt{\alpha_t} y_0, (1 - \alpha_t) \idm)$$ where $\alpha_t = \prod_{k\leq t} (1 - \beta_t)$.

  \pause
  It can be obtained from our process by defining $Y_{t} = X_{t} / \sqrt{\sigma_t^2 + 1}$. 


  Furthermore, knowing the score of $X_t$ can give access to the score of $Y_{t}$! (The two are equivalent).

\end{frame}
\begin{frame}{Backward Sampling (Variance preserving)}
  Algorithm:
  \begin{enumerate}
    \item Sample $\state_T \sim \gauss(0, \idm)$,
    \item Sample for $t = T-1, \cdots, 1$,
  \end{enumerate}
  $$\state_{t} \sim \gauss\left(\frac{1}{\sqrt{1-\beta_{t+1}}}(\state_{t+1} - \frac{\beta_{t+1}}{\sqrt{1 - \alpha_t}} \varepsilon_{\param}(\state_{t+1}, \sqrt{\alpha_{t+1}})),  \frac{\sigma_t^2}{\sigma_{t+1}^2}(\sigma_{t+1}^2 - \sigma_{t}^2)\idm\right) \eqsp,$$
  where the \textbf{epsilon net} $\varepsilon_{\param}$ can be linked to our denoining network by:
  $$D_{\param}(\lstate, \sigma(\alpha_t)) = \frac{1}{\sqrt{\beta_t}}\left(\lstate - \frac{\beta_t}{\sqrt{1 - \alpha_t}} \varepsilon_{\param}(\state_{t+1}, \sqrt{\alpha_t})\right) \eqsp,$$
  where $\sigma(\alpha_t) = \sqrt{\frac{1 - \alpha_t}{\alpha_t}}$.
\end{frame}
\begin{frame}{Diffusion posterior sampling \citep{chung2023diffusion}.}
    Remember our inverse problem posterior:
    \begin{equation}
      p_{0}(\lstate | y) = \frac{\ell(y | \lstate) p_{0}(\lstate)}{\int \ell(y | \tilde{\lstate}) p_{0}(\tilde{\lstate}) \rmd \tilde{\lstate}} \eqsp.
    \end{equation}
    \pause

    Can we use a diffusion model to sample from $p_{0}(\lstate | y)$ ?
\end{frame}
\begin{frame}{Diffused posteriors}
    \vspace{-100pt}
    We start by defining
    \begin{equation}
      p_{t}(\lstate | y) = \int p_{0}(\lstate_{0} | y) \gauss(\lstate; \lstate_{0}, \sigma_t^2 \idm) \rmd \lstate_{0}\eqsp.
    \end{equation}
    
    What is the score?
\end{frame}
\begin{frame}{DPS approximation \citep{chung2023diffusion}}

  \cite{chung2023diffusion} proposes the following approximation

  \begin{equation}
   \nabla \log p_{t}(\lstate | y) = \nabla \log p_{t}(\lstate) + \zeta_{t}\nabla \log l(y | \underbrace{\PE{}{\state_0 | \state_t = \lstate_t}}_{\lstate + \sigma^2 \nabla \log p_{t}(\lstate)})
  \end{equation}

  \pause
  In practice, for linear problems (i.e.) $\log l(y | x) = -\frac{1}{2\sigma_y^2} \| y - Ax \|^2 + c$, the authors suggests setting $\zeta_{t} = \tilde{\zeta} \frac{2\sigma_y^2}{ \| y - Ax \|}$ where $\tilde{\zeta}$ is a tunable constant (typically $1$ or $0.1$).
\end{frame}
